function startStickFigureAnimation() { clearInterval(animationInterval); animationFrameCount = 0; const stickArm = document.getElementById('stick-arm'); const stickForearm = document.getElementById('stick-forearm'); const stickBodyGroup = document.getElementById('stick-body-group'); animationInterval = setInterval(() => { if(circuitPaused || circuitResting || store.circuitRound > 3) return; animationFrameCount++; const currentEx = CONFIG.workoutCircuit[store.circuitExIdx]; const type = currentEx ? currentEx.type : 'bicep'; // Standard smooth loop for heavy movements const progress = (Math.sin(animationFrameCount * 0.15) + 1) / 2; // Higher frequency loop for rapid wrist burn const wristProgress = (Math.sin(animationFrameCount * 0.30) + 1) / 2; // Reset body positioning to neutral baseline stickBodyGroup.style.transform = 'rotate(0deg) translate(0px, 0px)'; if (type === 'bicep' || type === 'hammer') { const armRotation = -20 + (progress * 15); const forearmRotation = -85 + (progress * 80); stickArm.style.transform = `rotate(${armRotation}deg)`; stickForearm.style.transform = `rotate(${forearmRotation}deg)`; } else if (type === 'tricep') { const armRotation = -110 + (progress * 40); const forearmRotation = -50 - (progress * 50); stickArm.style.transform = `rotate(${armRotation}deg)`; stickForearm.style.transform = `rotate(${forearmRotation}deg)`; } else if (type === 'wrist_curl') { // Anchor upper arm, rapidly pump the forearm upwards stickArm.style.transform = 'rotate(10deg)'; const forearmRotation = -70 + (wristProgress * 25); stickForearm.style.transform = `rotate(${forearmRotation}deg)`; } else if (type === 'wrist_ext') { // Anchor upper arm, rapidly pump the forearm downwards/backwards stickArm.style.transform = 'rotate(10deg)'; const forearmRotation = -45 - (wristProgress * 25); stickForearm.style.transform = `rotate(${forearmRotation}deg)`; } }, 50); }
